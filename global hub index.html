Scholarship Scraper Template & JSON Bulk Importer

Use this Python script template to scrape university pages, or inject custom scholarship JSON arrays dynamically into your local browser database.

Python Scraper Template (BeautifulSoup & Requests) 
import requests
from bs4 import BeautifulSoup
import json

# Target University / Portal Scholarship Directory URL
URL = "https://example-university.edu/scholarships"

headers = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
}

def scrape_scholarships():
    response = requests.get(URL, headers=headers)
    if response.status_code != 200:
        print("Failed to reach website")
        return []

    soup = BeautifulSoup(response.text, "html.parser")
    scholarships = []

    for item in soup.select(".scholarship-card"):
        title = item.select_one(".title").get_text(strip=True) if item.select_one(".title") else "N/A"
        deadline = item.select_one(".deadline").get_text(strip=True) if item.select_one(".deadline") else "2026-12-31"
        link = item.select_one("a")["href"] if item.select_one("a") else URL

        scholarships.append({
            "title": title,
            "country": "Germany",
            "region": "Europe",
            "level": "Masters",
            "funding": "Fully Funded",
            "deadline": deadline,
            "status": "OPEN",
            "coverage": ["Tuition", "Stipend", "Airfare"],
            "officialLink": link,
            "summary": "Scraped program details directly from portal."
        })

    return json.dumps(scholarships, indent=2)

if __name__ == "__main__":
    print(scrape_scholarships())
